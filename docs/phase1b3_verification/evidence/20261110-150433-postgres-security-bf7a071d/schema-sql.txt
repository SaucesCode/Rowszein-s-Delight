BEGIN;
--
-- Create model AuthThrottleBucket
--
CREATE TABLE "accounts_auththrottlebucket" ("id" bigint NOT NULL PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY, "scope" varchar(2) NOT NULL, "key_version" smallint NOT NULL CHECK ("key_version" >= 0), "subject_digest" varchar(64) NOT NULL, "event_times" jsonb NOT NULL, "updated_at" timestamp with time zone NOT NULL);
--
-- Create model DeviceSession
--
CREATE TABLE "accounts_devicesession" ("id" uuid NOT NULL PRIMARY KEY, "user_epoch" bigint NOT NULL CHECK ("user_epoch" >= 0), "created_at" timestamp with time zone NOT NULL, "last_activity_at" timestamp with time zone NOT NULL, "absolute_expires_at" timestamp with time zone NOT NULL, "refresh_generation" bigint NOT NULL CHECK ("refresh_generation" >= 0), "refresh_jti_hash" varchar(64) NOT NULL, "revoked_at" timestamp with time zone NULL, "revocation_reason" varchar(24) NOT NULL);
--
-- Add field session_epoch to user
--
ALTER TABLE "accounts_user" ADD COLUMN "session_epoch" bigint DEFAULT 1 NOT NULL CHECK ("session_epoch" >= 0);
--
-- Create constraint acct_session_epoch_pos on model user
--
ALTER TABLE "accounts_user" ADD CONSTRAINT "acct_session_epoch_pos" CHECK (("session_epoch" >= 1 AND "session_epoch" <= 9223372036854775807));
--
-- Create index at_cleanup_time on field(s) updated_at of model auththrottlebucket
--
CREATE INDEX "at_cleanup_time" ON "accounts_auththrottlebucket" ("updated_at");
--
-- Create constraint at_scope_valid on model auththrottlebucket
--
ALTER TABLE "accounts_auththrottlebucket" ADD CONSTRAINT "at_scope_valid" CHECK ("scope" IN ('LA', 'LI', 'LR', 'RR'));
--
-- Create constraint at_keyver_pos on model auththrottlebucket
--
ALTER TABLE "accounts_auththrottlebucket" ADD CONSTRAINT "at_keyver_pos" CHECK (("key_version" >= 1 AND "key_version" <= 32767));
--
-- Create constraint at_digest_hex on model auththrottlebucket
--
ALTER TABLE "accounts_auththrottlebucket" ADD CONSTRAINT "at_digest_hex" CHECK ("subject_digest"::text ~ '\A[0-9a-f]{64}\Z');
--
-- Create constraint at_events_shape on model auththrottlebucket
--
ALTER TABLE "accounts_auththrottlebucket" ADD CONSTRAINT "at_events_shape" CHECK ((CASE WHEN jsonb_typeof("event_times") = 'array' THEN jsonb_array_length("event_times") <= 1000 ELSE false END));
--
-- Create constraint at_scope_key_unique on model auththrottlebucket
--
ALTER TABLE "accounts_auththrottlebucket" ADD CONSTRAINT "at_scope_key_unique" UNIQUE ("scope", "key_version", "subject_digest");
--
-- Add field user to devicesession
--
ALTER TABLE "accounts_devicesession" ADD COLUMN "user_id" bigint NOT NULL CONSTRAINT "accounts_devicesession_user_id_58ed1ceb_fk_accounts_user_id" REFERENCES "accounts_user"("id") DEFERRABLE INITIALLY DEFERRED; SET CONSTRAINTS "accounts_devicesession_user_id_58ed1ceb_fk_accounts_user_id" IMMEDIATE;
--
-- Create index ds_user_created on field(s) user, -created_at of model devicesession
--
CREATE INDEX "ds_user_created" ON "accounts_devicesession" ("user_id", "created_at" DESC);
--
-- Create index ds_abs_expiry on field(s) absolute_expires_at of model devicesession
--
CREATE INDEX "ds_abs_expiry" ON "accounts_devicesession" ("absolute_expires_at");
--
-- Create index ds_active_idle on field(s) last_activity_at of model devicesession
--
CREATE INDEX "ds_active_idle" ON "accounts_devicesession" ("last_activity_at") WHERE "revoked_at" IS NULL;
--
-- Create constraint ds_epoch_pos on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_epoch_pos" CHECK (("user_epoch" >= 1 AND "user_epoch" <= 9223372036854775807));
--
-- Create constraint ds_gen_nonneg on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_gen_nonneg" CHECK (("refresh_generation" >= 0 AND "refresh_generation" <= 9223372036854775807));
--
-- Create constraint ds_time_bounds on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_time_bounds" CHECK (("last_activity_at" >= ("created_at") AND "last_activity_at" < ("absolute_expires_at")));
--
-- Create constraint ds_abs_8h on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_abs_8h" CHECK ("absolute_expires_at" = ("created_at" + '0 days 28800.000000 seconds'::interval));
--
-- Create constraint ds_jti_hex on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_jti_hex" CHECK ("refresh_jti_hash"::text ~ '\A[0-9a-f]{64}\Z');
--
-- Create constraint ds_revocation_pair on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_revocation_pair" CHECK ((("revocation_reason" = '' AND "revoked_at" IS NULL) OR ("revocation_reason" IN ('logout', 'credentials_changed', 'role_changed', 'status_changed', 'refresh_reuse', 'idle_expired', 'absolute_expired', 'security_cutover', 'operator_revoked') AND "revoked_at" IS NOT NULL)));
--
-- Create constraint ds_revoked_after_start on model devicesession
--
ALTER TABLE "accounts_devicesession" ADD CONSTRAINT "ds_revoked_after_start" CHECK (("revoked_at" IS NULL OR "revoked_at" >= ("created_at")));
COMMIT;
